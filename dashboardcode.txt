function initials(name) { return name.split(' ').map(n=>n[0]).join('').slice(0,2).toUpperCase(); }
function avatarColor(name) { return avatarColors[name.charCodeAt(0) % avatarColors.length]; }

function statusBadge(s) {
  const map = {
    ready: ['status-ready','Ready'],
    washing: ['status-washing','Washing'],
    pending: ['status-pending','Pending'],
    picked: ['status-picked','Picked Up'],
    cancelled: ['status-cancelled','Cancelled'],
  };
  const [cls, label] = map[s] || ['status-pending', s];
  return `<span class="status-badge ${cls}">${label}</span>`;
}

// ─── LIVE WEIGHT SIMULATION ───────────────────────────────────────────────────
let liveW = 4.28;
let weightInterval;

function startWeightSim() {
  weightInterval = setInterval(() => {
    const delta = (Math.random() - 0.5) * 0.06;
    liveW = Math.max(0, Math.min(50, liveW + delta));
    const display = liveW.toFixed(2);
    const el1 = document.getElementById('live-weight');
    const el2 = document.getElementById('gauge-weight');
    if (el1) el1.textContent = display;
    if (el2) el2.textContent = display;

    // Update gauge arc
    const fill = document.getElementById('gauge-fill');
    if (fill) {
      const pct = liveW / 50;
      const circ = 565.5;
      const dash = circ * 0.75; // 270deg arc
      fill.style.strokeDasharray = circ;
      fill.style.strokeDashoffset = circ - (dash * pct);
    }

    // Update scale arc
    const arc = document.getElementById('scale-arc');
    if (arc) {
      const pct = liveW / 50;
      const deg = pct * 270;
      arc.style.transform = `rotate(${-45 + deg}deg)`;
    }
  }, 500);
}

document.getElementById('form-weight').addEventListener('input', updateCost);

function updateCost() {
  const w = parseFloat(document.getElementById('form-weight').value) || 0;
  const s = document.getElementById('form-service').value;
  document.getElementById('form-cost').textContent = '₱' + cost(w, s).toFixed(0);
}

function submitOrder() {
  const c = document.getElementById('form-customer').value.trim();
  const w = parseFloat(document.getElementById('form-weight').value);
  const s = document.getElementById('form-service').value;
  if (!c) { showToast('⚠️', 'Please enter customer name.'); return; }
  if (!w) { showToast('⚠️', 'Please enter weight.'); return; }
  const newOrder = {
    id: '#LM-' + (2041 + ORDERS.length + 1),
    customer: c, weight: w, service: s, status: 'pending',
    time: new Date().toLocaleTimeString('en-US',{hour:'2-digit',minute:'2-digit'})
  };
  ORDERS.unshift(newOrder);
  renderOrders('orders-body', ORDERS.slice(0,10));
  renderOrders('orders-body-full', ORDERS);
  const badge = document.getElementById('orders-badge');
  badge.textContent = parseInt(badge.textContent||0) + 1;
  closeModal();
  showToast('✅', `Order ${newOrder.id} created for ${c}.`);
}

function updateStatus(id) {
  const order = ORDERS.find(o => o.id === id);
  if (!order) return;
  const cycle = { pending:'washing', washing:'ready', ready:'picked', picked:'picked', cancelled:'cancelled' };
  order.status = cycle[order.status] || 'pending';
  renderOrders('orders-body', ORDERS.slice(0,10));
  renderOrders('orders-body-full', ORDERS);
  showToast('✅', `Order ${id} updated to ${order.status}.`);
}
